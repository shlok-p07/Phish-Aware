# numpy and pandas are split by interpreter rather than pinned once. The exact
# versions below 3.13 are the ones the awareness model was trained against, and
# CI pins that interpreter (see .python-version), so CI and production install
# byte-identical scientific stacks.
#
# Neither of those versions publishes a wheel for 3.13, and building them from
# source fails, so on a newer interpreter pip could not install this file at all
# -- which is why the suite had never been run on a machine with 3.13 and why the
# checked-in venv was missing joblib. The 3.13 line is the nearest release that
# does publish wheels. scikit-learn is not split: it is the version that
# unpickles the model artifact, and that pin has to hold everywhere.
numpy==2.0.2; python_version < "3.13"
numpy==2.1.3; python_version >= "3.13"
pandas==2.2.2; python_version < "3.13"
pandas==2.2.3; python_version >= "3.13"
